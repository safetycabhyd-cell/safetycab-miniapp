function bookRide() { let pickup = document.getElementById("pickup").value; let drop = document.getElementById("drop").value; if (pickup === "" || drop === "") { alert("Please enter Pick-up and Drop Location."); return; } window.open( "https://www.google.com/maps/dir/?api=1&origin=" + encodeURIComponent(pickup) + "&destination=" + encodeURIComponent(drop), "_blank" ); }
