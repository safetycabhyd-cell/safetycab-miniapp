// server.js const express = require("express"); const mysql = require("mysql2"); const app = express(); app.use(express.json()); // Database Connection const db = mysql.createConnection({ host: "localhost", user: "root", password: "password", database: "driver_app" }); db.connect(err => { if(err) throw err; console.log("Database Connected"); }); // 1. Driver Registration + First Location Save app.post("/add-driver", (req,res)=>{ const { driver_id, name, phone, vehicle_number, latitude, longitude } = req.body; const sql = ` INSERT INTO drivers (driver_id,name,phone,vehicle_number,latitude,longitude,online_status) VALUES (?,?,?,?,?,?,true) `; db.query(sql, [ driver_id, name, phone, vehicle_number, latitude, longitude ], (err,result)=>{ if(err) return res.json(err); res.json({ message:"Driver Added Successfully" }); }); }); // 2. Driver GPS Update app.post("/update-location",(req,res)=>{ const { driver_id, latitude, longitude, online_status } = req.body; const sql = ` UPDATE drivers SET latitude=?, longitude=?, online_status=?, last_update_time=NOW() WHERE driver_id=? `; db.query(sql, [ latitude, longitude, online_status, driver_id ], (err,result)=>{ if(err) return res.json(err); res.json({ message:"Location Updated" }); }); }); app.listen(3000,()=>{ console.log("Server Running on Port 3000"); });
