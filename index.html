const express = require("express"); const app = express(); app.use(express.json()); let drivers = []; // Add Driver app.post("/add-driver", (req, res) => { let driver = { driver_id: req.body.driver_id, name: req.body.name, phone: req.body.phone, vehicle_number: req.body.vehicle_number, latitude: req.body.latitude, longitude: req.body.longitude, online_status: true, last_update_time: new Date() }; drivers.push(driver); res.send(driver); }); // Update Location app.post("/update-location", (req,res)=>{ let driver = drivers.find( x => x.driver_id === req.body.driver_id ); if(!driver){ return res.status(404).send({ message:"Driver Not Found" }); } driver.latitude = req.body.latitude; driver.longitude = req.body.longitude; driver.last_update_time = new Date(); res.send(driver); }); // Show All Drivers app.get("/drivers",(req,res)=>{ res.json(drivers); }); app.listen(3000,()=>{ console.log("Server Running Port 3000"); });
