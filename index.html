// server.js const express = require("express"); const app = express(); app.use(express.json()); let drivers = []; // Driver Add app.post("/add-driver", (req, res) => { const driver = { driver_id: req.body.driver_id, name: req.body.name, phone: req.body.phone, vehicle_number: req.body.vehicle_number, latitude: req.body.latitude, longitude: req.body.longitude, online_status: true, last_update_time: new Date() }; drivers.push(driver); res.json({ message: "Driver Added", data: driver }); }); // GPS Update app.post("/update-location", (req,res)=>{ let driver = drivers.find( d => d.driver_id == req.body.driver_id ); if(!driver){ return res.json({ message:"Driver Not Found" }); } driver.latitude = req.body.latitude; driver.longitude = req.body.longitude; driver.online_status = true; driver.last_update_time = new Date(); res.json({ message:"Location Updated", data:driver }); }); // All Driver List app.get("/drivers",(req,res)=>{ res.json(drivers); }); app.listen(3000,()=>{ console.log("Mini App Server Started"); });
